async (page) => {
  // Run with playwright-cli run-code --filename against the local production page.
  // Actual encoded playback checks; intentionally separate from unit tests.
  const base = new URL(page.url()).origin;
  // Remove only this harness's previous fault route if an interrupted run
  // stopped before cleanup.
  await page.unroute('**/films/*.mp4');
  const results = [],
    errors = [],
    external = [];
  const check = (name, passed, detail = "") => {
    results.push({ name, passed, detail });
    if (!passed) throw new Error(name + ": " + JSON.stringify(detail));
  };
  let faultInjection = false;
  page.on("pageerror", (error) => errors.push(error.message));
  page.on("console", (message) => {
    if (message.type() === "error" && !faultInjection)
      errors.push(message.text());
  });
  page.on("request", (request) => {
    const url = request.url();
    if (url.startsWith("http") && new URL(url).origin !== base)
      external.push(url);
  });
  const ready = () =>
    page.waitForFunction(() => {
      const video = document.querySelector("video");
      const portrait = innerWidth <= innerHeight;
      return video?.readyState >= 2 && video.duration === 35
        && video.currentSrc.includes(portrait ? 'portrait' : 'landscape')
        && video.videoWidth === (portrait ? 1080 : 1920);
    });
  const state = () =>
    page.locator("video").evaluate((v) => ({
      time: v.currentTime,
      duration: v.duration,
      paused: v.paused,
      muted: v.muted,
      width: v.videoWidth,
      height: v.videoHeight,
      volume: v.volume,
      src: v.currentSrc,
      ended: v.ended,
      audioBytes: v.webkitAudioDecodedByteCount,
    }));
  const seek = async (time) => {
    await page.locator("video").evaluate((v, time) => {
      v.pause();
      v.currentTime = time;
    }, time);
    await page.waitForFunction((time) => {
      const v = document.querySelector("video");
      return !v.seeking && Math.abs(v.currentTime - time) < 0.05;
    }, time);
  };

  await page.emulateMedia({ reducedMotion: "no-preference" });
  await page.setViewportSize({ width: 1440, height: 900 });
  await page.goto(base + "/");
  await page.evaluate(() => window.scrollTo({ top: 0, behavior: 'instant' }));
  await ready();
  await page.waitForFunction(
    () => document.querySelector("video").currentTime > 0.2,
  );
  let value = await state();
  check(
    "Encoded landscape autoplays muted",
    value.width === 1920 &&
      value.height === 1080 &&
      value.muted &&
      !value.paused,
    value,
  );
  check(
    "Film opens at the top of the first viewport",
    (await page.locator(".film-intro").boundingBox()).y === 0,
  );
  check(
    "Slide transport and chapter strip removed",
    (await page
      .locator(".showcase-chapters, .showcase-transport, .pin-spacer")
      .count()) === 0,
  );
  await page.getByRole("button", { name: "Pause film", exact: true }).click();
  value = await state();
  await page.waitForTimeout(250);
  check(
    "User pause holds the decoded film",
    (await state()).paused &&
      Math.abs((await state()).time - value.time) < 0.05,
  );
  await seek(7);
  await page
    .getByRole("button", { name: "Sound off: watch with sound", exact: true })
    .click();
  await page.waitForFunction(
    () => document.querySelector("video").currentTime > 0.25,
  );
  value = await state();
  check(
    "First sound activation restarts with audible volume",
    !value.muted && !value.paused && value.volume === 0.9 && value.time < 3,
    value,
  );
  const signal = await page.evaluate(async () => {
    const video = document.querySelector("video");
    const context = new AudioContext();
    const source = context.createMediaElementSource(video);
    const analyser = context.createAnalyser();
    analyser.fftSize = 2048;
    source.connect(analyser);
    analyser.connect(context.destination);
    await context.resume();
    const samples = new Float32Array(analyser.fftSize);
    let maxRms = 0;
    for (let i = 0; i < 18; i++) {
      await new Promise((resolve) => setTimeout(resolve, 60));
      analyser.getFloatTimeDomainData(samples);
      maxRms = Math.max(
        maxRms,
        Math.sqrt(samples.reduce((sum, n) => sum + n * n, 0) / samples.length),
      );
    }
    window.__filmAudioCheck = { context, source, analyser };
    return {
      maxRms,
      context: context.state,
      decodedBytes: video.webkitAudioDecodedByteCount,
    };
  });
  check(
    "Browser decodes and routes a nonzero soundtrack",
    signal.context === "running" &&
      signal.maxRms > 0.005 &&
      signal.decodedBytes > 0,
    signal,
  );
  await page
    .getByRole("button", { name: "Sound on: mute film", exact: true })
    .click();
  check("Mute control mutes the native soundtrack", (await state()).muted);
  await page
    .getByRole("button", { name: "Show captions", exact: true })
    .click();
  await page.waitForFunction(
    () => document.querySelector("video").textTracks[0]?.cues?.length === 7,
  );
  check(
    "All seven English narration cues can be shown",
    await page
      .locator("video")
      .evaluate((v) => v.textTracks[0].mode === "showing"),
  );
  await page
    .getByRole("button", { name: "Hide captions", exact: true })
    .click();
  check(
    "Captions can be hidden",
    await page
      .locator("video")
      .evaluate((v) => v.textTracks[0].mode === "hidden"),
  );

  // A fresh media element avoids leaving the analyser attached to the player.
  await page.reload();
  await ready();
  await page.getByRole("button", { name: "Pause film", exact: true }).click();
  await seek(17);
  await page.setViewportSize({ width: 390, height: 844 });
  await ready();
  value = await state();
  check(
    "Portrait switch preserves user pause and timestamp",
    value.paused &&
      Math.abs(value.time - 17) < 0.1 &&
      value.width === 1080 &&
      value.height === 1920,
    value,
  );
  await page.setViewportSize({ width: 1440, height: 900 });
  await ready();
  value = await state();
  check(
    "Landscape switch preserves user pause and timestamp",
    value.paused && Math.abs(value.time - 17) < 0.1 && value.width === 1920,
    value,
  );
  await page.getByRole("button", { name: "Play film", exact: true }).click();
  await page.setViewportSize({ width: 390, height: 844 });
  await ready();
  await page.waitForFunction(() => !document.querySelector("video").paused);
  check(
    "Playing continues through a format switch",
    (await state()).time >= 17,
  );
  await page.getByRole("button", { name: "Pause film", exact: true }).click();
  await seek(34.7);
  await page.getByRole("button", { name: "Play film", exact: true }).click();
  await page.waitForFunction(() => document.querySelector("video").ended);
  await page.waitForTimeout(250);
  check(
    "Ending holds without looping and offers entry",
    (await state()).time === 35 &&
      (await page.getByRole("link", { name: "Find what’s next" }).isVisible()),
  );
  await page.setViewportSize({ width: 1440, height: 900 });
  await ready();
  value = await state();
  check(
    "Completed film stays completed across formats",
    value.paused &&
      value.time === 35 &&
      (await page.getByRole("button", { name: "Replay film" }).isVisible()),
    value,
  );
  await page.getByRole("button", { name: "Replay film", exact: true }).click();
  await page.waitForFunction(
    () =>
      !document.querySelector("video").paused &&
      document.querySelector("video").currentTime < 2,
  );
  check("Replay restarts the film", true);
  // A short viewport allows the following section to move the film entirely
  // offscreen; a 900px viewport still shows its bottom quarter at max scroll.
  await page.setViewportSize({ width: 1440, height: 600 });
  await page.evaluate(() =>
    window.scrollTo({ top: innerHeight + 100, behavior: "instant" }),
  );
  await page.waitForFunction(() => document.querySelector("video").paused);
  check("Offscreen playback pauses", true);
  await page.evaluate(() => window.scrollTo({ top: 0, behavior: "instant" }));
  await page.waitForFunction(() => !document.querySelector("video").paused);
  check("Visible playback resumes when user intent was playing", true);

  for (const [width, height] of [
    [320, 900],
    [390, 844],
    [768, 900],
    [1024, 900],
    [1440, 768],
    [700, 390],
  ]) {
    await page.setViewportSize({ width, height });
    await ready();
    if (!(await state()).paused)
      await page
        .getByRole("button", { name: "Pause film", exact: true })
        .click();
    await seek(17);
    const layout = await page.evaluate(() => {
      const controls = document
        .querySelector(".film-controls")
        .getBoundingClientRect();
      const entry = document
        .querySelector(".film-entry")
        .getBoundingClientRect();
      const stage = document
        .querySelector(".film-intro")
        .getBoundingClientRect();
      return {
        overflow: document.documentElement.scrollWidth > innerWidth,
        controls: {
          x: controls.x,
          right: controls.right,
          y: controls.y,
          bottom: controls.bottom,
        },
        entry: {
          x: entry.x,
          right: entry.right,
          y: entry.y,
          bottom: entry.bottom,
        },
        stageHeight: stage.height,
        fit: getComputedStyle(document.querySelector("video")).objectFit,
        format: document.querySelector(".film-intro").dataset.format,
      };
    });
    check(
      `Whole-frame layout and top controls at ${width}x${height}`,
      !layout.overflow &&
        layout.fit === "contain" &&
        layout.controls.x >= 0 &&
        layout.controls.right <= width &&
        layout.controls.bottom < height / 2 &&
        layout.entry.x >= 0 &&
        layout.entry.right <= width &&
        layout.entry.bottom < height / 2 &&
        Math.abs(layout.stageHeight - height) < 2,
      layout,
    );
    await page.screenshot({
      path: `output/playwright/film-${width}x${height}.png`,
    });
  }
  await page.setViewportSize({ width: 1440, height: 900 });
  await ready();
  for (const time of [3, 7, 12, 17, 22, 28, 34]) {
    await seek(time);
    await page.screenshot({
      path: `output/playwright/film-encoded-${time}.png`,
    });
  }
  await page
    .getByRole("link", { name: "Enter Opporbyte", exact: true })
    .click();
  await page.waitForURL("**/onboarding");
  check(
    "Entry opens the existing frontend preview",
    (await page.locator(".preview-label").innerText()) ===
      "FRONTEND PREVIEW · NO ACCOUNT REQUIRED",
  );
  await page.getByRole("button", { name: "Design", exact: true }).click();
  check(
    "Existing design filter works",
    (await page.locator("article").count()) === 2,
  );
  await page
    .getByRole("button", { name: "Save Product Designer example", exact: true })
    .click();
  check(
    "Existing save stays in the local preview",
    (await page
      .getByRole("button", { name: "Unsave Product Designer example" })
      .getAttribute("aria-pressed")) === "true",
  );
  await page
    .getByRole("textbox", { name: "Another field or role?" })
    .fill("architecture");
  check(
    "Existing empty state is honest",
    (await page.locator("article").count()) === 0 &&
      (await page.locator(".result-note").innerText()).includes(
        "No fictional examples",
      ),
  );
  await page.reload();
  check(
    "Existing preview selections reset on reload",
    (await page.locator(".result-note").innerText()).includes("0 saved"),
  );

  await page.emulateMedia({ reducedMotion: "reduce" });
  await page.goto(base + "/");
  await ready();
  value = await state();
  await page.waitForTimeout(400);
  check(
    "Reduced motion starts paused with immediate product entry",
    value.paused &&
      (await state()).time === value.time &&
      (await page.getByRole("link", { name: "Enter Opporbyte" }).isVisible()),
    value,
  );
  await page.getByRole("button", { name: "Play film", exact: true }).click();
  await page.waitForFunction(
    () => document.querySelector("video").currentTime > 0.2,
  );
  check("Reduced motion still permits explicit playback", true);
  await page.screenshot({ path: "output/playwright/film-reduced-motion.png" });

  faultInjection = true;
  await page.route("**/films/*.mp4", (route) => route.abort("failed"));
  await page.reload();
  await page
    .getByRole("status")
    .filter({ hasText: "The film couldn’t load" })
    .waitFor();
  check(
    "Media failure leaves the entry usable",
    await page.getByRole("link", { name: "Enter Opporbyte" }).isVisible(),
  );
  await page.screenshot({ path: "output/playwright/film-load-fallback.png" });
  await page
    .getByRole("link", { name: "Enter Opporbyte", exact: true })
    .click();
  await page.waitForURL("**/onboarding");
  check("Entry works after a failed movie load", true);
  await page.unroute("**/films/*.mp4");
  faultInjection = false;
  check(
    "No unexpected browser runtime or resource errors",
    errors.length === 0,
    errors,
  );
  check(
    "No external service requests in the frontend",
    external.length === 0,
    external,
  );
  await page.emulateMedia({ reducedMotion: "no-preference" });
  await page.goto(base + "/");
  await ready();
  return {
    base,
    checkedAt: new Date().toISOString(),
    results,
    errors,
    external,
    audioVerification: signal,
    perceptualListening: "not claimed",
  };
}
