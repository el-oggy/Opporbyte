async (page) => {
  const origin = new URL(page.url()).origin;
  const health = await page.request.get(origin + '/api/v1/health');
  if ((await health.json()).environment !== 'frontend-browser-qa') throw new Error('Refusing mutations outside isolated QA');
  const checks = [];
  const verify = (condition, label) => { if (!condition) throw new Error(label); checks.push(label); };
  const errors = [];
  const onError = error => errors.push(error.message);
  page.on('pageerror', onError);
  await page.context().clearCookies();
  await page.setViewportSize({ width: 1440, height: 960 });
  await page.goto(origin + '/login?next=/app/discover');
  await page.getByLabel('Email address', { exact: true }).fill('frontend-fixture@example.com');
  await page.getByLabel('Password', { exact: true }).fill('isolated-test-owner-password');
  await page.getByRole('button', { name: 'Sign in to workspace', exact: true }).click();
  await page.waitForURL('**/app/discover');
  await page.getByLabel('Configured source', { exact: true }).waitFor();
  await page.getByRole('link', { name: /Review role/ }).first().waitFor();
  verify(await page.getByLabel('Active career profile').inputValue() === 'software', 'Existing software profile selected');
  verify(await page.getByRole('button', { name: 'Sync now', exact: true }).isDisabled(), 'Unconfigured source cannot sync');
  await page.getByRole('button', { name: 'Connect a job board', exact: true }).click();
  await page.getByLabel('Employer board slug', { exact: true }).fill('https://invalid.example/board');
  verify(await page.getByLabel('Employer board slug', { exact: true }).evaluate(e => e.validity.patternMismatch), 'Board input rejects full URLs before submission');
  await page.getByLabel('Employer board slug', { exact: true }).fill('fixture-board');
  const firstResponse = page.waitForResponse(r => r.url().endsWith('/automation/syncs') && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Configure and sync', exact: true }).click();
  const response = await firstResponse;
  const task = await response.json();
  const firstRequest = response.request().postDataJSON();
  verify(response.ok() && task.status === 'partial', 'Configured public source produces persisted partial status');
  verify(task.result.new_jobs_count === 1 && task.result.duplicates_skipped_count === 1, 'Duplicate source records produce one canonical job');
  verify(task.result.rejected_count === 1 && task.result.eligible_count === 1, 'Malformed record isolated and eligible job ranked');
  await page.getByRole('dialog').getByRole('status').filter({ hasText: 'Sync completed' }).waitFor();
  await page.keyboard.press('Escape');
  verify(await page.getByRole('button', { name: 'Connect a job board', exact: true }).evaluate(e => e === document.activeElement), 'Connection drawer returns keyboard focus');
  const panel = page.getByRole('region', { name: 'Job source sync' });
  await panel.getByText('partial', { exact: true }).waitFor();
  verify((await panel.innerText()).includes('malformed record'), 'Partial failure explained in persistent history');
  await page.getByRole('link', { name: 'Fixture Discovery Engineer', exact: true }).waitFor();
  const jobCard = page.locator('article').filter({ has: page.getByRole('link', { name: 'Fixture Discovery Engineer', exact: true }) });
  verify((await jobCard.innerText()).includes('% match'), 'Existing Discover card displays matching evidence score');
  await jobCard.getByRole('button', { name: 'Save to queue', exact: true }).click();
  await jobCard.getByRole('button', { name: 'In queue', exact: true }).waitFor();
  await page.reload();
  await jobCard.getByRole('button', { name: 'In queue', exact: true }).waitFor();
  verify(await jobCard.getByRole('button', { name: 'In queue', exact: true }).isDisabled(), 'Saved role remains in review queue after reload');
  const persisted = await page.request.get(origin + '/api/v1/applications');
  const application = (await persisted.json()).find(a => a.job_title === 'Fixture Discovery Engineer');
  verify(application?.status === 'ready_for_review' && !application.submitted_at, 'Sync and save do not submit an application');
  await page.screenshot({ path: 'output/playwright/phase5/discover-desktop.png', fullPage: true });
  const replay = await page.request.post(origin + '/api/v1/automation/syncs', { data: firstRequest });
  verify((await replay.json()).id === task.id, 'Identical request key returns original persisted task');
  const sourceControls = await (await page.request.get(origin + '/api/v1/automation/sources')).json();
  const nextRequestAt = sourceControls.find(source => source.source_id === task.parameters.source_id).next_request_at;
  if (nextRequestAt) await page.waitForTimeout(Math.max(0, Date.parse(nextRequestAt) - Date.now()) + 100);
  const repeatResponse = page.waitForResponse(r => r.url().endsWith('/automation/syncs') && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Sync now', exact: true }).click();
  const repeated = await (await repeatResponse).json();
  verify(repeated.result.new_jobs_count === 0, 'Fresh sync safely reuses canonical job');
  await page.getByRole('button', { name: 'Connect a job board', exact: true }).click();
  await page.getByLabel('Employer board slug', { exact: true }).fill('fixture-failure');
  const failureResponse = page.waitForResponse(r => r.url().endsWith('/automation/syncs') && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Configure and sync', exact: true }).click();
  const failed = await (await failureResponse).json();
  verify(failed.status === 'failed' && failed.result.attempts === 3, 'Transient source failure stops after bounded retries');
  await page.getByRole('dialog').getByRole('alert').waitFor();
  verify((await page.getByRole('dialog').getByRole('alert').innerText()).includes('http_503'), 'Source failure shown without a false success message');
  await page.keyboard.press('Escape');
  await panel.getByText('failed', { exact: true }).waitFor();
  await page.screenshot({ path: 'output/playwright/phase5/failure-desktop.png', fullPage: true });
  await page.getByRole('button', { name: 'Connect a job board', exact: true }).click();
  await page.getByLabel('Employer board slug', { exact: true }).fill('fixture-slow');
  const cancelResponse = page.waitForResponse(r => r.url().endsWith('/automation/syncs') && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Configure and sync', exact: true }).click();
  await page.keyboard.press('Escape');
  await panel.getByRole('button', { name: 'Cancel sync', exact: true }).click();
  const cancelled = await (await cancelResponse).json();
  verify(cancelled.status === 'cancelled', 'Running sync cooperatively cancels before persistence');
  verify((await panel.innerText()).includes('cancelled'), 'Cancellation remains visible in sync history');
  await page.getByLabel('Configured source', { exact: true }).selectOption(task.parameters.source_id);
  const excludedUpdate = await page.request.put(origin + '/api/v1/profiles/software', { data: { excluded_companies: ['fixture-board'] } });
  verify(excludedUpdate.ok(), 'Existing profile preference API controls exclusions');
  const exclusionResponse = page.waitForResponse(r => r.url().endsWith('/automation/syncs') && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Sync now', exact: true }).click();
  const excluded = await (await exclusionResponse).json();
  verify(excluded.result.eligible_count === 0 && excluded.result.excluded_count === 1, 'Excluded employer is filtered before ranking');
  await page.getByRole('link', { name: 'Fixture Discovery Engineer', exact: true }).waitFor({ state: 'detached' });
  checks.push('Excluded employer absent from profile Discover results');
  await page.request.put(origin + '/api/v1/profiles/software', { data: { excluded_companies: [] } });
  await page.reload();
  await page.getByRole('link', { name: 'Fixture Discovery Engineer', exact: true }).waitFor();
  await page.getByLabel('Configured source', { exact: true }).locator('option').filter({ hasText: 'fixture-board' }).waitFor({ state: 'attached' });
  for (const width of [320, 390, 1024]) {
    await page.setViewportSize({ width, height: 900 });
    await panel.getByLabel('Configured source', { exact: true }).waitFor();
    if (width < 1000) await page.waitForFunction(() => document.querySelector('#workspace-navigation').getBoundingClientRect().right <= 1);
    verify(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth + 1), `Discover sync has no overflow at ${width}px`);
    await page.screenshot({ path: `output/playwright/phase5/discover-${width}.png`, fullPage: true });
  }
  await page.setViewportSize({ width: 1440, height: 960 });
  await page.goto(origin + '/app/jobs/' + encodeURIComponent(task.result.job_ids[0]));
  await page.getByRole('heading', { name: 'Fixture Discovery Engineer', exact: true }).waitFor();
  checks.push('Synced opportunity uses existing job review screen');
  const apiRequests = [];
  const collect = r => { if (r.url().includes('/api/v1/')) apiRequests.push(r.url()); };
  await page.goto(origin + '/preview/discover');
  await page.getByRole('button', { name: 'Connect a job board', exact: true }).waitFor();
  page.on('request', collect);
  await page.getByRole('button', { name: 'Connect a job board', exact: true }).click();
  await page.getByLabel('Employer board slug', { exact: true }).fill('fixture-board');
  await page.getByRole('button', { name: 'Try preview', exact: true }).click();
  await page.getByRole('dialog').getByRole('status').filter({ hasText: 'Preview only' }).waitFor();
  verify(apiRequests.length === 0, 'Preview connection makes no API request or external sync');
  page.off('request', collect); page.off('pageerror', onError);
  verify(errors.length === 0, 'No JavaScript page errors during sync workflow');
  return { passed: checks.length, checks, externalSourcesContacted: 0, fixtureTaskId: task.id };
}
